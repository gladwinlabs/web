How we work

No surprises, no slide decks

01

Agree on what good looks like

Before we start, we write down what the software has to do and how fast it has to do it. Then we check against that at the end.

02

Show working software every week

You see it running on a staging site, not described in a deck.

03

Leave it in good shape

Docs, tests and setup notes, so your own team can take over whenever you want.

deploy/release-check.ts
import { probe, notify } from "./ops";

export async function checkRelease(env: Env) {
  const results = await Promise.all([
    probe(env.api,   { p95: 250, unit: "ms" }),
    probe(env.db,    { replicaLag: 0 }),
    probe(env.queue, { backlog: 0 }),
  ]);

  if (results.every(r => r.ok)) return "promote";

  await notify("release held", results);
  return "rollback"; // nobody gets woken up at 3am
}
